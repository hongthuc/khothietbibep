    // Render toàn bộ các section sản phẩm
    function renderAllProductSections(filterCategory = 'all', searchKeyword = '') {
      const kw = searchKeyword.trim().toLowerCase();
      const filterBar = document.getElementById('active-filter-bar');
      const filterNameEl = document.getElementById('active-filter-name');

      if (filterCategory !== 'all' || kw !== '') {
        filterBar.classList.remove('hidden');
        filterBar.classList.add('flex');
        filterNameEl.textContent = kw ? `"${searchKeyword}"` : filterCategory;
      } else {
        filterBar.classList.add('hidden');
        filterBar.classList.remove('flex');
      }

      const sectionsMap = {
        'combo': document.getElementById('grid-combo'),
        'bep-tu': document.getElementById('grid-bep-tu'),
        'hut-mui': document.getElementById('grid-hut-mui'),
        'chau-voi': document.getElementById('grid-chau-voi'),
        'rua-chen': document.getElementById('grid-rua-chen'),
        'phu-kien': document.getElementById('grid-phu-kien')
      };

      Object.keys(sectionsMap).forEach(secKey => {
        const gridEl = sectionsMap[secKey];
        if (!gridEl) return;
        const parentSec = gridEl.closest('.product-section');

        const filtered = PRODUCTS_DATA.filter(item => {
          const matchSec = item.section === secKey;
          const matchCat = filterCategory === 'all' || item.category.toLowerCase().includes(filterCategory.toLowerCase());
          const matchKw = !kw || item.name.toLowerCase().includes(kw) || item.category.toLowerCase().includes(kw);
          return matchSec && matchCat && matchKw;
        });

        if (filtered.length === 0 && (filterCategory !== 'all' || kw !== '')) {
          parentSec.classList.add('hidden');
        } else {
          parentSec.classList.remove('hidden');
          gridEl.innerHTML = filtered.map(createProductCardHTML).join('');
        }
      });
    }

    function showBannerSlide(index) {
      currentBannerIndex = (index + 2) % 2;
      const slide0 = document.getElementById('banner-slide-0');
      const slide1 = document.getElementById('banner-slide-1');
      const dot0 = document.getElementById('banner-dot-0');
      const dot1 = document.getElementById('banner-dot-1');

      if (currentBannerIndex === 0) {

        slide0.classList.add('block');
        slide1.classList.add('hidden');
        slide1.classList.remove('block');
        dot0.className = 'w-6 h-2 rounded-full bg-amber-400 transition-all';
        dot1.className = 'w-2 h-2 rounded-full bg-white/60 transition-all';
      } else {
        slide1.classList.remove('hidden');
        slide1.classList.add('block');
        slide0.classList.add('hidden');
        slide0.classList.remove('block');
        dot1.className = 'w-6 h-2 rounded-full bg-amber-400 transition-all';
        dot0.className = 'w-2 h-2 rounded-full bg-white/60 transition-all';
      }
    }

    }

    // Giỏ hàng & báo giá
    function addToCart(productId) {
      const p = PRODUCTS_DATA.find(item => item.id === productId);
      if (!p) return;
      const existing = cartItems.find(i => i.id === productId);
      if (existing) {
        existing.qty += 1;
      } else {
        cartItems.push({ ...p, qty: 1 });
      }
      updateCartBadge();
      openCartModal();
    }

    function removeFromCart(productId) {
      cartItems = cartItems.filter(i => i.id !== productId);
      updateCartBadge();
      renderCartModalItems();
    }

    function updateCartBadge() {
      const totalQty = cartItems.reduce((sum, item) => sum + item.qty, 0);
      document.getElementById('cart-count-badge').textContent = totalQty;
    }

    function renderCartModalItems() {
      const container = document.getElementById('cart-items-container');
      const totalEl = document.getElementById('cart-total-price');

      if (cartItems.length === 0) {
        container.innerHTML = `<p class="text-center text-slate-500 py-6">Chưa có sản phẩm nào trong danh sách báo giá.</p>`;
        totalEl.textContent = '0đ';
        return;
      }

      let sum = 0;
      container.innerHTML = cartItems.map(item => {
        sum += item.price * item.qty;
        return `
          <div class="flex items-center justify-between gap-2 p-2 rounded border border-slate-200 bg-white">
            <div class="flex-1">
              <p class="font-bold text-slate-800 leading-snug">${item.name}</p>
              <p class="text-red-600 font-bold mt-0.5">${formatVND(item.price)} x ${item.qty}</p>
            </div>
            <button type="button" onclick="removeFromCart('${item.id}')" class="text-slate-400 hover:text-red-600 p-1">
              ✕
            </button>
          </div>
        `;
      }).join('');

      totalEl.textContent = formatVND(sum);
    }

    function openCartModal() {
      renderCartModalItems();
      const modal = document.getElementById('cart-modal');
      modal.classList.remove('hidden');
      modal.classList.add('flex');
    }

    function closeCartModal() {
      const modal = document.getElementById('cart-modal');
      modal.classList.add('hidden');
      modal.classList.remove('flex');
    }
